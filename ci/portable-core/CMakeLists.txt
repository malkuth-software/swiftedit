cmake_minimum_required(VERSION 3.24)
project(SwiftEditPortableCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# This gate builds production CSV/formula code without the native SDK. It is
# deliberately not named or packaged as a SwiftEdit application build.
enable_testing()
add_executable(swiftedit-newline-tests ../../tests/newline_tests.cpp ../../src/newlines.cpp)
target_include_directories(swiftedit-newline-tests PRIVATE ../../src)
add_test(NAME newlines COMMAND swiftedit-newline-tests)
add_library(swiftedit_csv STATIC ../../src/csv.cpp)
target_include_directories(swiftedit_csv PUBLIC ../../src)
add_executable(swiftedit-csv-tests ../../tests/csv_tests.cpp)
target_link_libraries(swiftedit-csv-tests PRIVATE swiftedit_csv)
add_test(NAME csv COMMAND swiftedit-csv-tests)
if(WIN32)
  set(PAGED_SOURCE ../../src/paged_windows.cpp)
else()
  set(PAGED_SOURCE ../../src/paged_posix.cpp)
endif()
add_executable(swiftedit-paged-tests ../../tests/paged_tests.cpp ${PAGED_SOURCE})
target_include_directories(swiftedit-paged-tests PRIVATE ../../src)
target_compile_definitions(swiftedit-paged-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME paged-file COMMAND swiftedit-paged-tests)
set_tests_properties(paged-file PROPERTIES TIMEOUT 15)
if(WIN32)
  set(FILE_SOURCE ../../src/file_windows.cpp)
else()
  set(FILE_SOURCE ../../src/file_posix.cpp)
endif()
add_executable(swiftedit-file-tests ../../tests/file_adapter_tests.cpp ${FILE_SOURCE})
target_include_directories(swiftedit-file-tests PRIVATE ../../src)
target_compile_definitions(swiftedit-file-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN SWIFTEDIT_FILE_TESTING)
add_test(NAME file-adapter COMMAND swiftedit-file-tests)
set_tests_properties(file-adapter PROPERTIES TIMEOUT 15)
